Csucs <- function(v){
eredmeny <- c()
for(i in 2:(length(v)-1)){
if((v[i] > v[i-1])  & (v[i] > v[i+1]) ){
eredmeny <- c(eredmeny, v[i])
}
}
return(eredmeny)
}
Csucs(c(1, 5, 3, 7, 2, 4, 6, 1))✅Jó


Sorozat <- function(n){
if(n <= 0 | n %% 1 != 0){
print("A bemenet nem megfelelő!")
return()
}
sorozat <- c(n)

while(n != 1) {
if(n %% 2 == 0) {
n <- n / 2 
} else if((n %% 2 == 1 ) & ( n > 10) {
n <- n - 3 
} else {
n <- n - 1
}
sorozat <- c(sorozat, n)
}
return(sorozat)
}
